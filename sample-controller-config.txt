process.roles=controller

node.id=1

# list of node running on this node, internally it might be doing new ServerSocket(9093)
# open a network listener called 'CONTROLLER'(network door for other processes, other processes
# running on another JVM would want to connect this process and for that we need to open
# a network listener) on port 9093
listeners=CONTROLLER://:9093
# listeners=CLIENT://:9092,CONTROLLER://:9093,REPLICATION://:9094

# The listener named CONTROLLER(bind with 9093) is the listener Kafka uses for
# controller-to-controller communication.
controller.listener.names=CONTROLLER

# these controllers uses these port to connect to another controllers
controller.quorum.voters=1@localhost:9093,2@localhost:9193,3@localhost:9293

# PLAINTEXT (no encryption). In production, we should use SSL
# assume the CONTROLLER 1 received bytes from network, now kafka needs to intepret those bytes
# kafka uses protocol PLAINTEXT to understand the message. It's still would be in binary but there
# is no encryption/auth layer to decode that
listener.security.protocol.map=CONTROLLER:PLAINTEXT

# here all files will be persisted
log.dirs=/Users/bhaskarsharan/Desktop/open-source-data-tools/kafka/logs/controller-1

# Default partitions when creating a topic(if not specified during creation process)
num.partitions=3
default.replication.factor=2
